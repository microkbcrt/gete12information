
<!-- 开始：彩虹天气通知组件 -->
<style>
    /* 组件样式 */
    #rainbow-weather-widget {
        display: inline-block;
        position: relative;
        padding: 25px 50px; /* 调整内边距以适应大字号 */
        border-radius: 50px;
        font-family: -apple-system, BlinkMacSystemFont, "Microsoft YaHei", sans-serif;
        font-size: 36px; /* 修改为 36px */
        font-weight: 900; /* 加粗 */
        color: #ffffff;
        text-align: center;
        text-shadow: 0 2px 8px rgba(0,0,0,0.3);
        /* 彩虹渐变背景 */
        background: linear-gradient(90deg, #ff0000, #ff7300, #fffb00, #48ff00, #00ffd5, #002bff, #7a00ff, #ff00c8, #ff0000);
        background-size: 400% 400%;
        animation: rainbow-flow 6s ease infinite;
        box-shadow: 0 8px 20px rgba(0,0,0,0.25);
        cursor: default;
        white-space: nowrap;
        line-height: 1.5;
    }

    /* 渐变流动动画 */
    @keyframes rainbow-flow {
        0% { background-position: 0% 50%; }
        50% { background-position: 100% 50%; }
        100% { background-position: 0% 50%; }
    }

    /* 文字容器 */
    #rainbow-weather-widget .text-box {
        display: inline-block;
        opacity: 0;
        transform: translateY(10px);
        transition: all 0.8s ease;
    }

    /* 显示状态 */
    #rainbow-weather-widget .text-box.show {
        opacity: 1;
        transform: translateY(0);
    }
</style>

<div id="rainbow-weather-widget">
    <span class="text-box" id="rw-text-target"></span>
</div>

<script>
    (function() {
        // 配置内容
        const content = "3 月 5 日起天气逐渐转好";
        const target = document.getElementById('rw-text-target');
        const widget = document.getElementById('rainbow-weather-widget');
        
        if (!target || !widget) return;

        let i = 0;
        const speed = 100; // 打字速度

        // 打字机效果函数
        function typeWriter() {
            if (i < content.length) {
                target.innerHTML += content.charAt(i);
                i++;
                setTimeout(typeWriter, speed);
            } else {
                // 打字完成后添加显示类，触发上浮动画
                target.classList.add('show');
            }
        }

        // 延迟一点启动
        setTimeout(typeWriter, 300);
    })();
</script>
<!-- 结束：彩虹天气通知组件 -->
