  const cache = new Map(); // 记录 文本节点DOM -> 原始内容
  let timer = null;

  // 核心替换
  const replaceTxt = str => str.replace(/[\u4e00-\u9fa5]/g, c => DICT[c] || c);

  // Vue 环境安全过滤
  const SKIP_TAGS = /^(SCRIPT|STYLE|TEXTAREA|INPUT|SELECT|OPTION|CODE|PRE|NOSCRIPT|IFRAME|SVG|CANVAS|TEMPLATE|META|LINK|BR|IMG)$/i;

  function processNode(node) {
    const p = node.parentElement;
    if (!p) return;
    // 跳过 Vue 关键组件/表格/代码块
    if (SKIP_TAGS.test(p.tagName)) return;
    if (p.closest('[data-antonym-skip], .el-table, .vxe-table, [v-cloak]')) return;

    const curr = node.textContent;
    if (!curr) return;

    const orig = cache.get(node);
    if (orig === undefined) {
      // 首次遇到：缓存原文并替换
      cache.set(node, curr);
      node.textContent = replaceTxt(curr);
    } else if (curr === orig) {
      // Vue 恢复原文或新节点：重新替换
      node.textContent = replaceTxt(curr);
    } else if (curr !== replaceTxt(orig)) {
      // 网站/Ajax/Vue 动态修改了内容：更新缓存并替换
      cache.set(node, curr);
      node.textContent = replaceTxt(curr);
    }
    // 其他情况：已是替换状态，跳过
  }

  // 1秒扫描主逻辑
  function scan() {
    const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, null);
    let node;
    while ((node = walker.nextNode())) processNode(node);

    // 🧹 防内存泄漏：Vue SPA 路由切换会销毁旧节点，定期清理无效引用
    if (cache.size > 600) {
      for (const [n] of cache) {
        if (!document.body.contains(n)) cache.delete(n);
      }
    }
  }

  // 清理旧定时器（防云控重复注入堆积）
  if (timer) clearInterval(timer);
  scan(); // 立即执行一次
  timer = setInterval(scan, 1000);

  // 🌐 暴露复原接口（云控需要恢复时直接执行此字符串）
  window.反义字复原 = function() {
    clearInterval(timer);
    timer = null;
    for (const [node, orig] of cache) {
      if (document.body.contains(node)) node.textContent = orig;
    }
    cache.clear();
  };
})();
