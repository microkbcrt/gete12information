<script>
(function () {
  if (document.getElementById('__gaokao_firework__')) return;
  const wrapperId = '__gaokao_firework__';

  function init() {
    if (document.getElementById(wrapperId)) return;

    // ============ 全屏黑色覆盖层 ============
    const container = document.createElement('div');
    container.id = wrapperId;
    container.style.cssText = `
      position: fixed; top: 0; left: 0; width: 100vw; height: 100vh;
      z-index: 99999; pointer-events: none; overflow: hidden;
      background-color: #000000;  /* 纯黑背景 */
    `;
    document.body.appendChild(container);

    // ============ Canvas ============
    const canvas = document.createElement('canvas');
    canvas.style.cssText = `
      position: absolute; top: 0; left: 0; width: 100%; height: 100%;
      display: block;
    `;
    const ctx = canvas.getContext('2d');
    container.appendChild(canvas);

    // ============ 文字层 ============
    const textDiv = document.createElement('div');
    textDiv.textContent = '高考必胜';
    textDiv.style.cssText = `
      position: absolute; top: 50%; left: 50%;
      transform: translate(-50%, -50%) scale(0);
      font-size: clamp(2.8rem, 8vw, 6.5rem);
      font-weight: 900;
      color: #FFD700;
      letter-spacing: 0.1em;
      text-shadow:
        0 0 12px #FFA500,
        0 0 28px #FFD700,
        0 0 55px #FF8C00,
        0 0 100px #FFD700,
        0 0 150px #FFA500,
        0 0 220px #FFD700;
      white-space: nowrap;
      opacity: 0;
      will-change: transform, opacity;
    `;
    container.appendChild(textDiv);

    // ============ 响应式尺寸 ============
    function resize() {
      const dpr = Math.min(window.devicePixelRatio || 1, 2);
      canvas.width = window.innerWidth * dpr;
      canvas.height = window.innerHeight * dpr;
      canvas.style.width = window.innerWidth + 'px';
      canvas.style.height = window.innerHeight + 'px';
      ctx.setTransform(1, 0, 0, 1, 0, 0);
      ctx.scale(dpr, dpr);
    }
    resize();
    window.addEventListener('resize', resize);

    const W = () => window.innerWidth;
    const H = () => window.innerHeight;

    // ============ 工具函数 ============
    function rand(min, max) { return min + Math.random() * (max - min); }
    function randInt(min, max) { return Math.floor(rand(min, max + 1)); }

    const PALETTE = [
      '#FFD700', '#FFA500', '#FF8C00', '#FF4444', '#FF6B6B',
      '#FF3030', '#FF69B4', '#FF1493', '#FFEC8B', '#FFFFFF',
      '#FFFACD', '#FF6347', '#FF4500'
    ];
    function randomColor() { return PALETTE[randInt(0, PALETTE.length - 1)]; }
    function randomGold() {
      const g = ['#FFD700', '#FFC125', '#FFB90F', '#FFA500', '#FFEC8B', '#FFF8DC', '#FFE4B5'];
      return g[randInt(0, g.length - 1)];
    }

    // ============ 粒子类 ============
    class Particle {
      constructor(x, y, vx, vy, color, life, size) {
        this.x = x; this.y = y; this.vx = vx; this.vy = vy;
        this.color = color; this.life = life; this.maxLife = life;
        this.size = size || rand(2.5, 6);
        this.gravity = 0.06; this.friction = 0.985;
      }
      update() {
        this.vx *= this.friction; this.vy *= this.friction;
        this.vy += this.gravity;
        this.x += this.vx; this.y += this.vy;
        this.life--;
      }
      draw(ctx) {
        const alpha = Math.max(0, this.life / this.maxLife);
        const r = this.size * (0.5 + 0.5 * alpha);
        ctx.save();
        ctx.globalAlpha = alpha;
        ctx.fillStyle = this.color;
        ctx.shadowColor = this.color;
        ctx.shadowBlur = r * 4;
        ctx.beginPath();
        ctx.arc(this.x, this.y, r, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
      }
      get dead() { return this.life <= 0; }
    }

    // ============ 烟花火箭 ============
    class Firework {
      constructor() {
        this.x = rand(W() * 0.12, W() * 0.88);
        this.y = H() + 20;
        this.targetY = rand(H() * 0.1, H() * 0.45);
        this.vy = rand(-16, -11);
        this.vx = rand(-1.8, 1.8);
        this.color = randomColor();
        this.trail = [];
        this.exploded = false;
        this._particlesGenerated = false;
      }

      update() {
        if (this.exploded) return;
        this.y += this.vy;
        this.x += this.vx;
        if (Math.random() < 0.8) {
          this.trail.push(new Particle(
            this.x + rand(-3, 3), this.y + rand(4, 12),
            rand(-0.6, 0.6), rand(1.5, 3.5),
            this.color, randInt(15, 30), rand(2.5, 5)
          ));
        }
        if (this.y <= this.targetY) {
          this.exploded = true;
        }
      }

      generateExplosionParticles() {
        if (this._particlesGenerated) return [];
        this._particlesGenerated = true;
        const count = randInt(55, 100);
        const arr = [];
        for (let i = 0; i < count; i++) {
          const angle = rand(0, Math.PI * 2);
          const speed = rand(3.5, 11);
          arr.push(new Particle(
            this.x, this.y,
            Math.cos(angle) * speed, Math.sin(angle) * speed,
            Math.random() < 0.5 ? this.color : randomColor(),
            randInt(35, 80), rand(2, 6.5)
          ));
        }
        return arr;
      }

      draw(ctx) {
        if (this.exploded) return;
        ctx.save();
        ctx.fillStyle = '#FFFFFF';
        ctx.shadowColor = this.color;
        ctx.shadowBlur = 14;
        ctx.beginPath();
        ctx.arc(this.x, this.y, 4.5, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
      }

      get dead() {
        return this.exploded && this._particlesGenerated && this.trail.length === 0;
      }
    }

    // ============ 冲击波 ============
    class Shockwave {
      constructor(x, y, maxR, dur) {
        this.x = x; this.y = y; this.radius = 0; this.maxRadius = maxR;
        this.duration = dur; this.elapsed = 0; this.alive = true;
      }
      update(dt) {
        this.elapsed += dt;
        const p = this.elapsed / this.duration;
        this.radius = this.maxRadius * p;
        if (p >= 1) this.alive = false;
      }
      draw(ctx) {
        const p = this.elapsed / this.duration;
        const alpha = 0.65 * (1 - p);
        ctx.save();
        ctx.globalAlpha = alpha;
        ctx.strokeStyle = '#FFD700';
        ctx.shadowColor = '#FFD700';
        ctx.shadowBlur = 22;
        ctx.lineWidth = 10 * (1 - p);
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
        ctx.stroke();
        ctx.restore();
      }
      get dead() { return !this.alive; }
    }

    // ============ 全局状态 ============
    let fireworks = [];
    let particles = [];
    let shockwaves = [];
    let sparkles = [];
    let fireworkTimer = 0;
    let fireworkInterval = 0.28;
    let textShown = false;
    let centralBurstTimer = 0;
    const CENTRAL_BURST_INTERVAL = 4.8; // 每4.8秒一次中央爆发

    // 中央金色爆发
    function centralBurst() {
      const cx = W() / 2, cy = H() / 2;
      for (let i = 0; i < 90; i++) {
        const a = rand(0, Math.PI * 2);
        const s = rand(5, 15);
        particles.push(new Particle(cx, cy, Math.cos(a) * s, Math.sin(a) * s, randomGold(), randInt(30, 70), rand(2.5, 8)));
      }
      const maxR = Math.max(W(), H()) * 0.72;
      shockwaves.push(new Shockwave(cx, cy, maxR, 0.75));
      setTimeout(() => shockwaves.push(new Shockwave(cx, cy, maxR * 0.85, 0.55)), 100);
      setTimeout(() => shockwaves.push(new Shockwave(cx, cy, maxR * 0.7, 0.4)), 200);
    }

    function showText() {
      if (textShown) return;
      textShown = true;
      textDiv.style.opacity = '1';
      textDiv.style.animation = 'gaokaoBounceIn 0.75s cubic-bezier(0.175, 0.885, 0.32, 1.275) forwards';
      centralBurst();
    }

    // 注入弹跳动画
    const styleEl = document.createElement('style');
    styleEl.textContent = `
      @keyframes gaokaoBounceIn {
        0%   { transform: translate(-50%, -50%) scale(0); }
        35%  { transform: translate(-50%, -50%) scale(1.3); }
        55%  { transform: translate(-50%, -50%) scale(0.88); }
        72%  { transform: translate(-50%, -50%) scale(1.06); }
        85%  { transform: translate(-50%, -50%) scale(0.96); }
        94%  { transform: translate(-50%, -50%) scale(1.02); }
        100% { transform: translate(-50%, -50%) scale(1); }
      }
    `;
    document.head.appendChild(styleEl);

    // ============ 持续循环 ============
    let lastTime = performance.now();
    let sparkleTimer = 0;

    function loop(ts) {
      const dt = Math.min((ts - lastTime) / 1000, 0.1);
      lastTime = ts;

      // 文字定时出现（第1.1秒首次出现，之后保持可见不再隐藏）
      if (!textShown && (ts - performance.now() > 1100 || performance.now() === ts)) {
        // 使用时间差判断，实际首次显示用简单延时
      }
      // 更稳健的文字触发：利用起始时间
      if (!textShown && (ts - startTime) > 1100) {
        showText();
      }

      // 周期性中央爆发（文字出现后才开始计时）
      if (textShown) {
        centralBurstTimer += dt;
        if (centralBurstTimer >= CENTRAL_BURST_INTERVAL) {
          centralBurstTimer = 0;
          centralBurst();
        }
      }

      // 持续发射烟花
      fireworkTimer += dt;
      const interval = textShown ? rand(0.35, 0.65) : rand(0.22, 0.38);
      if (fireworkTimer >= fireworkInterval) {
        fireworkTimer = 0;
        fireworkInterval = interval;
        fireworks.push(new Firework());
      }
      // 确保始终有烟花
      if (fireworks.length === 0 && !textShown) {
        fireworks.push(new Firework());
      }

      // 文字周围光点
      if (textShown) {
        sparkleTimer += dt;
        if (sparkleTimer > 0.035) {
          sparkleTimer = 0;
          const cx = W() / 2, cy = H() / 2;
          sparkles.push(new Particle(
            cx + rand(-W() * 0.2, W() * 0.2), cy + rand(-H() * 0.07, H() * 0.07),
            rand(-0.9, 0.9), rand(-2.8, -1), randomGold(), randInt(45, 95), rand(1.2, 4)
          ));
          if (sparkles.length > 250) sparkles.splice(0, sparkles.length - 250);
        }
      }

      // 更新所有对象
      fireworks.forEach(f => f.update());
      particles.forEach(p => p.update());
      shockwaves.forEach(s => s.update(dt));
      sparkles.forEach(s => s.update());

      // 处理爆炸粒子生成
      fireworks.forEach(f => {
        if (f.exploded && !f._particlesGenerated) {
          particles.push(...f.generateExplosionParticles());
        }
      });

      // 更新尾焰
      fireworks.forEach(f => {
        f.trail.forEach(p => p.update());
        f.trail = f.trail.filter(p => !p.dead);
      });

      // 清理死亡对象
      fireworks = fireworks.filter(f => !f.dead);
      particles = particles.filter(p => !p.dead);
      shockwaves = shockwaves.filter(s => !s.dead);
      sparkles = sparkles.filter(s => !s.dead);

      // 性能限制
      const maxP = W() < 480 ? 350 : W() < 768 ? 550 : 800;
      if (particles.length > maxP) particles.splice(0, particles.length - maxP);

      // ===== 绘制黑色背景 + 烟花 =====
      ctx.clearRect(0, 0, W(), H());
      // 纯黑背景
      ctx.fillStyle = '#000000';
      ctx.fillRect(0, 0, W(), H());

      shockwaves.forEach(s => s.draw(ctx));
      fireworks.forEach(f => f.trail.forEach(p => p.draw(ctx)));
      fireworks.forEach(f => f.draw(ctx));
      particles.forEach(p => p.draw(ctx));
      sparkles.forEach(s => s.draw(ctx));

      requestAnimationFrame(loop);
    }

    const startTime = performance.now();
    // 保证首帧就有烟花发射
    fireworks.push(new Firework());
    requestAnimationFrame(loop);
  }

  if (document.body) {
    init();
  } else {
    document.addEventListener('DOMContentLoaded', init);
  }
})();
</script>
