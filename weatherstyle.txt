<script>
  (function () {
    var progress = 15;
    var root = document.getElementById('summer-progress');
    var bar = document.getElementById('summer-progress-bar');
    var text = document.getElementById('summer-progress-text');
    var isDark = window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches;

    function applyTheme(dark) {
      root.style.background = dark ? '#111827' : '#ffffff';
      root.style.color = dark ? '#f9fafb' : '#111827';
      root.style.border = dark ? '1px solid rgba(255,255,255,0.12)' : '1px solid rgba(0,0,0,0.08)';
      root.style.boxShadow = dark ? '0 4px 16px rgba(0,0,0,0.35)' : '0 4px 16px rgba(0,0,0,0.06)';
      root.querySelector('div > div + div').style.background = dark
        ? 'rgba(255,255,255,0.12)'
        : 'rgba(0,0,0,0.08)';
    }

    bar.style.width = progress + '%';
    text.textContent = progress + '%';
    applyTheme(isDark);

    if (window.matchMedia) {
      var media = window.matchMedia('(prefers-color-scheme: dark)');
      if (media.addEventListener) {
        media.addEventListener('change', function (e) {
          applyTheme(e.matches);
        });
      } else if (media.addListener) {
        media.addListener(function (e) {
          applyTheme(e.matches);
        });
      }
    }
  })();
</script>
