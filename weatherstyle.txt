OpenClaw，中文昵称“小龙虾”。自2026年初发布以来迅速走红，因其强大的自动化能力被广泛应用于个人办公、开发辅助和团队协作中，真正实现了“用自然语言替代人工完成重复性工作”
<script>
// 清除弹窗的通用函数
function removePopup() {
  // 移除弹窗元素
  const overlays = document.querySelectorAll('.overlay');
  overlays.forEach(el => el.remove());
  
  // 可选：同时移除相关的样式
  const styles = document.querySelectorAll('style');
  styles.forEach(style => {
    if (style.textContent.includes('.overlay')) {
      style.remove();
    }
  });
  
  console.log('弹窗已清除');
}

// 立即执行清除
removePopup();

// 或者将函数暴露到全局，方便其他地方调用
window.removePopup = removePopup;
</script>
